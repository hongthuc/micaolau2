        function renderProducts(list) {
            const grid = document.getElementById('mainProductGrid');
            const countText = document.getElementById('productCountText');
            if (!grid) return;

            if (list.length === 0) {
                grid.innerHTML = `<div class="col-span-full bg-white p-6 rounded-xl text-center text-xs sm:text-sm text-gray-600 border border-gray-200">Không tìm thấy sản phẩm phù hợp. <button onclick="filterCategory('all')" class="text-brand-700 font-bold underline ml-1">Xem tất cả sản phẩm</button></div>`;

                grid.innerHTML = list.map(createProductCardHTML).join('');
            }
            if (countText) {
                countText.textContent = `Hiển thị ${list.length} sản phẩm thực tế`;

        function filterCategory(catKey) {
            document.querySelectorAll('.cat-pill').forEach(btn => {
                if (btn.getAttribute('data-cat-btn') === catKey) {
                    btn.className = "cat-pill px-2.5 py-2 sm:py-1.5 rounded-md text-[11px] sm:text-xs font-semibold bg-brand-700 text-white shadow-sm transition text-center leading-tight";
                } else {
                    btn.className = "cat-pill px-2.5 py-2 sm:py-1.5 rounded-md text-[11px] sm:text-xs font-semibold bg-brand-50 text-brand-800 hover:bg-brand-100 border border-brand-200 transition text-center leading-tight";
                }
            });

            if (catKey === 'all') {
                renderProducts(productsData);
            } else {
                renderProducts(productsData.filter(p => p.category === catKey));
            }
        }

        function handleSearch(inputId) {
            const input = document.getElementById(inputId);
            if (!input) return;
            const keyword = input.value.trim().toLowerCase();
            if (!keyword) {
                renderProducts(productsData);
                return;
            }
            const filtered = productsData.filter(p =>
                p.name.toLowerCase().includes(keyword) ||
                p.categoryLabel.toLowerCase().includes(keyword) ||
                p.desc.toLowerCase().includes(keyword)
            );
            renderProducts(filtered);
            document.getElementById('san-pham-trang-chu').scrollIntoView({ behavior: 'smooth' });
        }

        let quoteCounter = 0;
        function selectProductForQuote(id) {
            const p = productsData.find(item => item.id === id);
            if (!p) return;
            quoteCounter++;
            const badge = document.getElementById('quoteCountBadge');
            if (badge) badge.textContent = quoteCounter;

            const selectEl = document.getElementById('qProduct');
            if (selectEl) {
                let matched = false;
                for (let i = 0; i < selectEl.options.length; i++) {
                    if (p.name.includes(selectEl.options[i].value)) {
                        selectEl.selectedIndex = i;
                        matched = true;
                        break;
                    }
                }
                if (!matched) {
                    const opt = document.createElement('option');
                    opt.value = p.name;
                    opt.textContent = p.name;
                    selectEl.appendChild(opt);
                    selectEl.value = p.name;
                }
            }
            document.getElementById('form-bao-gia').scrollIntoView({ behavior: 'smooth' });
            document.getElementById('qName').focus();
        }

        /* Xử lý Slider 2 Banner Tự Trượt */
        let currentSlideIndex = 0;
        const totalSlides = 2;
        let slideInterval;

        function showSlide(index) {
            const slides = document.querySelectorAll('.hero-slide');
            const dots = document.querySelectorAll('.slide-dot');
            slides.forEach((sl, idx) => {
                if (idx === index) {
                    sl.classList.remove('hidden', 'opacity-0');
                    sl.classList.add('opacity-100');
                } else {
                    sl.classList.add('hidden', 'opacity-0');
                    sl.classList.remove('opacity-100');
                }
            });
            dots.forEach((dot, idx) => {
                if (idx === index) {
                    dot.className = "slide-dot w-2.5 h-2.5 rounded-full bg-amber-400 transition";
                } else {
                    dot.className = "slide-dot w-2.5 h-2.5 rounded-full bg-white/50 transition";
                }
            });
            currentSlideIndex = index;
        }

        function changeSlide(step) {
            let next = (currentSlideIndex + step + totalSlides) % totalSlides;
            showSlide(next);
            resetSlideTimer();
        }

        function goToSlide(index) {
            showSlide(index);
            resetSlideTimer();
        }

        function resetSlideTimer() {
            clearInterval(slideInterval);
            slideInterval = setInterval(() => {
                changeSlide(1);
            }, 5000);
        }

        /* Hàm tự động quét ngược toàn bộ các thẻ cha của WordPress/Flatsome để xóa lề trái 15px gây lệch phải */
        function fixWordPressParentLayout() {
            const wrapper = document.getElementById('bonthinh-site-wrapper');
            if (!wrapper) return;
            let parent = wrapper.parentElement;
            while (parent && parent !== document.documentElement) {
                parent.style.setProperty('padding-left', '0px', 'important');
                parent.style.setProperty('padding-right', '0px', 'important');
                parent.style.setProperty('margin-left', '0px', 'important');
                parent.style.setProperty('margin-right', '0px', 'important');
                parent.style.setProperty('width', '100%', 'important');
                parent.style.setProperty('max-width', '100%', 'important');
                parent.style.setProperty('min-width', '0px', 'important');
                parent.style.setProperty('left', '0px', 'important');
                parent.style.setProperty('right', '0px', 'important');
                parent.style.setProperty('box-sizing', 'border-box', 'important');
                parent = parent.parentElement;
            }
            document.documentElement.style.setProperty('padding', '0px', 'important');
            document.documentElement.style.setProperty('margin', '0px', 'important');
            document.documentElement.style.setProperty('overflow-x', 'hidden', 'important');
            if (document.body) {
                document.body.style.setProperty('padding', '0px', 'important');
                document.body.style.setProperty('margin', '0px', 'important');
                document.body.style.setProperty('overflow-x', 'hidden', 'important');
            }
        }
